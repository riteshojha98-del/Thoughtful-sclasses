const MAX_FILE_BYTES = 4 * 1024 * 1024;

function uid() {
  return Date.now().toString(36) + Math.random().toString(36).slice(2, 8);
}

// Extract Google Drive file ID from link
function extractDriveFileId(url) {
  if (!url) return null;
  const match = url.match(/(?:drive\.google\.com.*[?&]id=|yourdomain\.com\/file\/)([a-zA-Z0-9-_]+)/);
  return match ? match[1] : null;
}

// Get preview URL from Drive
function getDrivePreviewUrl(fileId) {
  return `https://drive.google.com/file/d/${fileId}/preview`;
}

function FileBadge({ type }) {
  const isPdf = type === "application/pdf";
  return (
    <div className={`file-badge ${isPdf ? "file-badge-pdf" : "file-badge-img"}`}>
      {isPdf ? "PDF" : type === "drive" ? "📁" : "IMG"}
    </div>
  );
}

function Timer({ seconds, isActive }) {
  const [timeLeft, setTimeLeft] = useState(seconds);

  useEffect(() => {
    if (!isActive) return;
    if (timeLeft <= 0) {
      setTimeLeft(0);
      return;
    }

    const interval = setInterval(() => {
      setTimeLeft(prev => prev - 1);
    }, 1000);

    return () => clearInterval(interval);
  }, [timeLeft, isActive]);

  const minutes = Math.floor(timeLeft / 60);
  const secs = timeLeft % 60;
  const isLow = timeLeft <= 10;

  return (
    <div className={`timer ${isLow ? "timer-warning" : ""}`}>
      {minutes}:{secs.toString().padStart(2, "0")}
    </div>
  );
}

    e.preventDefault();

      setError("Please fill all fields");
      return;

      return;

      role,
      email,
      name: name || email.split("@")[0],

          {!isLogin && (

              <input
                className="text-input"
                placeholder="e.g. Ananya Singh"
                value={name}
                onChange={(e) => setName(e.target.value)}
              />
            </>

            className="text-input"
            placeholder="your@email.com"

/* ====== ASSIGNMENT LIST ====== */

function AssignmentList({ editable, isTeacher }) {
  const [assignments, setAssignments] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadAssignments();
  }, []);

  const loadAssignments = async () => {
    try {
      const result = await window.storage?.get("assignments:list");
      if (result) {
        setAssignments(JSON.parse(result.value));
      }
    } catch (err) {
      console.error("Load error:", err);
    }
    setLoading(false);
  };

  const deleteAssignment = async (id) => {
    const updated = assignments.filter(a => a.id !== id);
    setAssignments(updated);
    await window.storage?.set("assignments:list", JSON.stringify(updated));
  };

  if (loading) return <div className="loading-line">Loading...</div>;

  return (
    <div>
      {assignments.length === 0 ? (
        <EmptyBoard label={isTeacher ? "No assignments posted yet" : "No assignments available"} />
      ) : (
        <div className="assignment-grid">
          {assignments.map(assignment => (
            <div key={assignment.id} className="assignment-card">
              <div className="assignment-card-top">
                <h3 className="assignment-title">{assignment.title}</h3>
                {editable && (
                  <button className="link-btn link-btn-danger" onClick={() => deleteAssignment(assignment.id)}>
                    ✕
                  </button>
                )}
              </div>
              <p className="assignment-desc">{assignment.description}</p>

              {assignment.driveLink && (
                <div className="file-link">
                  <FileBadge type="drive" />
                  <a href={assignment.driveLink} target="_blank" rel="noopener noreferrer" className="link-btn">
                    View on Drive
                  </a>
                </div>
              )}

              {assignment.dueDate && (
                <span className="due-pill">Due: {assignment.dueDate}</span>
              )}
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

/* ====== POST ASSIGNMENT ====== */

function PostAssignment({ onPosted, setError }) {
  const [title, setTitle] = useState("");
  const [description, setDescription] = useState("");
  const [driveLink, setDriveLink] = useState("");
  const [dueDate, setDueDate] = useState("");
  const [loading, setLoading] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");

    if (!title.trim()) {
      setError("Title is required");
      return;
    }

    setLoading(true);

    try {
      const assignment = {
        id: uid(),
        title: title.trim(),
        description: description.trim(),
        driveLink: driveLink.trim(),
        dueDate: dueDate,
        createdAt: new Date().toISOString(),
      };

      // Get existing assignments
      let assignments = [];
      const result = await window.storage?.get("assignments:list");
      if (result) {
        assignments = JSON.parse(result.value);
      }

      // Add new assignment
      assignments.push(assignment);
      await window.storage?.set("assignments:list", JSON.stringify(assignments));

      setTitle("");
      setDescription("");
      setDriveLink("");
      setDueDate("");
      onPosted();
    } catch (err) {
      setError("Failed to post assignment");
    }
    setLoading(false);
  };

  return (
    <div className="board-card">
      <h3 className="font-hand text-xl text-chalk mb-4">Post New Assignment</h3>

      <form onSubmit={handleSubmit}>
        <label className="field-label">Title *</label>
        <input
          className="text-input"
          placeholder="e.g. Chapter 5 - Algebra Exercises"
          value={title}
          onChange={(e) => setTitle(e.target.value)}
          required
        />

        <label className="field-label" style={{ marginTop: "1rem" }}>Description</label>
        <textarea
          className="text-input"
          placeholder="Add instructions or details..."
          value={description}
          onChange={(e) => setDescription(e.target.value)}
          rows={3}
          style={{ resize: "vertical" }}
        />

        <label className="field-label" style={{ marginTop: "1rem" }}>Google Drive Link</label>
        <input
          className="text-input"
          placeholder="Paste shared Google Drive link"
          value={driveLink}
          onChange={(e) => setDriveLink(e.target.value)}
        />
        <p className="field-hint">Share Google Drive file/folder with students. Link must be shareable.</p>

        <label className="field-label" style={{ marginTop: "1rem" }}>Due Date</label>
        <input
          type="date"
          className="text-input"
          value={dueDate}
          onChange={(e) => setDueDate(e.target.value)}
        />

        <button type="submit" className="primary-btn" style={{ marginTop: "1rem" }} disabled={loading}>
          {loading ? "Posting..." : "Post Assignment"}
        </button>
      </form>
    </div>
  );
}

/* ====== TEACHER MCQ QUIZ ====== */

function TeacherQuiz({ setError }) {
  const [quiz, setQuiz] = useState(null);
  const [question, setQuestion] = useState("");
  const [options, setOptions] = useState(["", "", "", ""]);
  const [correct, setCorrect] = useState(0);
  const [loading, setLoading] = useState(false);

  const startQuiz = async (e) => {
    e.preventDefault();
    setError("");

    if (!question.trim() || options.some(o => !o.trim())) {
      setError("Question and all options required");
      return;
    }

    setLoading(true);

    try {
      const newQuiz = {
        id: uid(),
        question: question.trim(),
        options: options.map(o => o.trim()),
        correctIndex: correct,
        status: "live",
        responses: [],
        revealed: false,
        createdAt: new Date().toISOString(),
      };

      await window.storage?.set(`quiz:${newQuiz.id}`, JSON.stringify(newQuiz));
      await window.storage?.set("quiz:active", newQuiz.id);

      setQuiz(newQuiz);
      setQuestion("");
      setOptions(["", "", "", ""]);
      setCorrect(0);
    } catch (err) {
      setError("Failed to start quiz");
    }
    setLoading(false);
  };

  const revealAnswers = async () => {
    const updated = { ...quiz, revealed: true };
    setQuiz(updated);
    await window.storage?.set(`quiz:${quiz.id}`, JSON.stringify(updated));
  };

  const endQuiz = async () => {
    const updated = { ...quiz, status: "ended" };
    await window.storage?.set(`quiz:${quiz.id}`, JSON.stringify(updated));
    await window.storage?.delete("quiz:active");
    setQuiz(null);
  };

  if (quiz && quiz.status === "live") {
    return (
      <div className="board-card">
        <div className="live-badge-row">
          <span className="live-badge">● LIVE</span>
          <span className="response-count">{quiz.responses.length} responses</span>
        </div>

        <h3 className="font-hand text-lg text-chalk mb-3">{quiz.question}</h3>

        <div className="results-list">
          {quiz.options.map((option, idx) => {
            const count = quiz.responses.filter(r => r === idx).length;
            const percent = quiz.responses.length ? ((count / quiz.responses.length) * 100) : 0;
            const isCorrect = idx === quiz.correctIndex;

            return (
              <div key={idx} className={isCorrect ? "result-row-correct" : ""}>
                <div className="result-row-label">
                  <span>{option}</span>
                  <span>{count} ({Math.round(percent)}%)</span>
                </div>
                <div className="result-bar-track">
                  <div className="result-bar-fill" style={{ width: `${percent}%` }} />
                </div>
              </div>
            );
          })}
        </div>

        <div className="quiz-actions">
          {!quiz.revealed && (
            <button className="primary-btn" onClick={revealAnswers}>
              Reveal Answer
            </button>
          )}
          <button className="link-btn link-btn-danger" onClick={endQuiz}>
            End Quiz
          </button>
        </div>
      </div>
    );
  }

  return (
    <div className="board-card">
      <h3 className="font-hand text-xl text-chalk mb-4">Create Live MCQ Quiz</h3>

      <form onSubmit={startQuiz}>
        <label className="field-label">Question *</label>
        <textarea
          className="text-input"
          placeholder="Write your question here..."
          value={question}
          onChange={(e) => setQuestion(e.target.value)}
          rows={2}
        />

        <label className="field-label" style={{ marginTop: "1rem" }}>Options *</label>
        {options.map((opt, idx) => (
          <div key={idx} style={{ marginBottom: "0.5rem" }}>
            <div style={{ display: "flex", gap: "0.5rem", alignItems: "center" }}>
              <input
                type="radio"
                name="correct"
                checked={correct === idx}
                onChange={() => setCorrect(idx)}
                className="option-radio"
              />
              <input
                className="text-input"
                placeholder={`Option ${String.fromCharCode(65 + idx)}`}
                value={opt}
                onChange={(e) => {
                  const newOpts = [...options];
                  newOpts[idx] = e.target.value;
                  setOptions(newOpts);
                }}
              />
            </div>
          </div>
        ))}

        <button type="submit" className="primary-btn" style={{ marginTop: "1rem" }} disabled={loading}>
          {loading ? "Starting..." : "Start Quiz"}
        </button>
      </form>
    </div>
  );
}

/* ====== STUDENT MCQ QUIZ ====== */

function StudentQuiz({ setError }) {
  const [quiz, setQuiz] = useState(null);
  const [myChoice, setMyChoice] = useState(null);
  const [timeLimit, setTimeLimit] = useState(60);
  const [quizActive, setQuizActive] = useState(false);

  useEffect(() => {
    const checkQuiz = async () => {
      try {
        const active = await window.storage?.get("quiz:active");
        if (active) {
          const quizData = await window.storage?.get(`quiz:${active.value}`);
          if (quizData) {
            setQuiz(JSON.parse(quizData.value));
            setQuizActive(true);
            setMyChoice(null);
          }
        }
      } catch (err) {
        console.error("Quiz fetch error:", err);
      }
    };

    const interval = setInterval(checkQuiz, 2000);
    checkQuiz();
    return () => clearInterval(interval);
  }, []);

  const submitAnswer = async (index) => {
    if (myChoice !== null) return;

    setMyChoice(index);

    // Save response
    const updated = { ...quiz };
    updated.responses.push(index);
    await window.storage?.set(`quiz:${quiz.id}`, JSON.stringify(updated));
  };

  if (!quizActive || !quiz) {
    return (
      <EmptyBoard label="No live quiz right now. Wait for teacher to start one..." />
    );
  }

  return (
    <div className="board-card">
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: "1rem" }}>
        <span className="live-badge">● LIVE</span>
        <Timer seconds={timeLimit} isActive={!quiz.revealed} />
      </div>

      <h3 className="font-hand text-lg text-chalk mb-4">{quiz.question}</h3>

      <div className="mcq-options">
        {quiz.options.map((option, idx) => {
          const isSelected = myChoice === idx;
          const isCorrect = idx === quiz.correctIndex;
          const showCorrect = quiz.revealed && isCorrect;

          return (
            <button
              key={idx}
              className={`mcq-option ${isSelected ? "mcq-option-picked" : ""} ${showCorrect ? "mcq-option-correct" : ""}`}
              onClick={() => submitAnswer(idx)}
              disabled={myChoice !== null || !quizActive}
            >
              {option}
              {showCorrect && <ChalkCircle className="chalk-answer-circle" />}
            </button>
          );
        })}
      </div>

      {myChoice !== null && !quiz.revealed && (
        <p className="waiting-line">✓ Answer recorded — waiting for teacher to reveal...</p>
      )}

      {quiz.revealed && (
        <p className={myChoice === quiz.correctIndex ? "form-success" : "form-error"}>
          {myChoice === quiz.correctIndex
            ? "🎉 Correct! Well done!"
            : `❌ The correct answer is: ${quiz.options[quiz.correctIndex]}`}
        </p>
      )}
    </div>
  );
}
